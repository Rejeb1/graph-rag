# This file is what the Dockerfile installs — keep it slim, production-only.
# The Ragas/LangChain evaluation stack lives in requirements-eval.txt +
# requirements-ragas.txt instead: it's only used by scripts/run_benchmark.py,
# never by the deployed API, and pulling it into the image roughly doubles
# build time/size for nothing.

# CPU-only PyTorch build. sentence-transformers depends on torch but doesn't
# pin it, so plain PyPI resolution grabs the full CUDA build (several GB of
# nvidia-*/cuda-toolkit wheels) even on a CPU-only host like Render's free
# tier. Pinning torch here, before sentence-transformers, makes pip satisfy
# sentence-transformers' unpinned "torch" requirement with this CPU wheel
# instead. (Harmless on Windows dev machines — the CUDA-bundling behavior
# this works around is Linux-wheel-specific.)
--extra-index-url https://download.pytorch.org/whl/cpu
torch>=2.0

# LLM provider — Groq only (free, no card required). See src/config.py.
